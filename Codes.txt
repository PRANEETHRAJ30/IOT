const int smokePin = A0;
const int soilPin = A1;

int smokeValue = 0;
int soilValue = 0;

void setup() {
  Serial.begin(9600);

  Serial.println("Smoke and Soil Moisture Monitoring");
}

void loop() {
  // Read sensor values
  smokeValue = analogRead(smokePin);
  soilValue = analogRead(soilPin);

  // Display smoke sensor value
  Serial.print("Smoke Value: ");
  Serial.println(smokeValue);

  // Display soil moisture sensor value
  Serial.print("Soil Moisture Value: ");
  Serial.println(soilValue);

  // Check smoke level
  if (smokeValue > 300) {
    Serial.println("Smoke Detected!");
  } else {
    Serial.println("Smoke Level Normal");
  }

  // Check soil moisture condition
  if (soilValue > 600) {
    Serial.println("Soil is Dry");
  } else {
    Serial.println("Soil Moisture Normal");
  }

  Serial.println("------------------------");

  delay(1000);
}

{

{

  if (irValue == LOW)
  {
    Serial.println("Object Detected");

#include <DHT.h>

#define DHTPIN 2
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(9600);
  Serial.println(F("DHT TEST!"));
  dht.begin();
}

void loop() {
  delay(2000);

  float h = dht.readHumidity();
  float t = dht.readTemperature();

  if (isnan(h) || isnan(t)) {
    Serial.println(F("Failed to read from DHT sensor!"));
    return;
  }

  Serial.print(F("Humidity: "));
  Serial.print(h);
  Serial.print(F("%  Temperature: "));
  Serial.println(t);
}

#include <DHT.h>
#include <ESP8266WiFi.h>

String apiKey = "9872OV4AXQJ6918";       // Your ThingSpeak Write API Key
const char* ssid = "349";                 // WiFi SSID
const char* pass = "your_password";       // Missing in your notes: required for WiFi.begin()
const char* server = "api.thingspeak.com";

#define DHTPIN D3
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);
WiFiClient client;

void setup() {
  Serial.begin(115200);
  delay(1000);
  dht.begin();

  Serial.println("Connecting to ");
  Serial.println(ssid);

  WiFi.begin(ssid, pass);

  while (WiFi.status() != WL_CONNECTED) {
    delay(2000);
    Serial.println(".");
  }

  Serial.println("");
  Serial.println("WiFi connected");
}

void loop() {
  float h = dht.readHumidity();
  float t = dht.readTemperature();

  if (isnan(h) || isnan(t)) {
    Serial.println("Failed to read from DHT sensor!");
    return;
  }

  if (client.connect(server, 80)) {
    String postStr = apiKey;
    postStr += "&field1=";
    postStr += String(t);
    postStr += "&field2=";
    postStr += String(h);
    postStr += "\r\n\r\n";

    client.print("POST /update HTTP/1.1\n");
    client.print("Host: api.thingspeak.com\n");
    client.print("Connection: close\n");
    client.print("X-THINGSPEAKAPIKEY: " + apiKey + "\n");
    client.print("Content-Type: application/x-www-form-urlencoded\n");
    client.print("Content-Length: ");
    client.print(postStr.length());
    client.print("\n\n");
    client.print(postStr);

    Serial.print("Temperature: ");
    Serial.print(t);
    Serial.print(" degrees Celsius, Humidity: ");
    Serial.print(h);
    Serial.println("%. Send to Thingspeak.");
  }
  client.stop();

  Serial.println("Waiting...");
  // ThingSpeak needs at least a 15-second delay between updates
  delay(15000); 
}

#include <ESP8266WiFi.h>
#include <WiFiUdp.h>
#include <DHT.h>

const char* ssid = "349";
const char* password = "12341234";
const char* udpAddress = "172.16.105.144";
const int udpPort = 8080;

#define DHTPIN D3
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);
WiFiUDP udp;

void setup() {
  Serial.begin(115200);
  Serial.println();
  Serial.println("Connecting to WiFi: ");

  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.println("Connecting...");
  }

  Serial.println();
  Serial.print("Connected to WiFi IP: ");
  Serial.println(WiFi.localIP());

  dht.begin();
}

void loop() {
  delay(1000);

  float temperature = dht.readTemperature();
  float humidity = dht.readHumidity();

  if (isnan(temperature) || isnan(humidity)) {
    Serial.println("Failed to read from DHT Sensor!");
    return;
  }

  Serial.print("Temperature: ");
  Serial.print(temperature);
  Serial.print(" Humidity: ");
  Serial.println(humidity);

  Serial.println("Sending data over UDP: ");

  udp.beginPacket(udpAddress, udpPort);
  udp.print("Temp: ");
  udp.print(temperature);
  udp.print(" Humidity: ");
  udp.print(humidity);
  udp.print("\n");
  udp.endPacket();

  Serial.println("Data sent over UDP!");
}
